    from emp e

    group by e.name 
    order by total_work_exp desc;
  counter number := 0; 
begin 
  for i in work loop
    counter := counter + 1;
    dbms_output.put_line('name: '||i.name);

    dbms_output.put_line('-----------------------------');

    end if;
