SQL> CREATE OR REPLACE FUNCTION phone(phone_code IN VARCHAR2)
  2  RETURN VARCHAR2
  3  IS
  4      code VARCHAR2(255) := phone_code || '%';  -- Concatenate '%' for LIKE pattern
  5      names VARCHAR2(4000) := '';  -- Initialize names as an empty string (Oracle doesn't like ' ' as initial value)
  6
  7      CURSOR clientphone IS
  8          SELECT name
  9          FROM client
 10          WHERE phone LIKE code;
 11  BEGIN
 12      -- Loop through the records in the clientphone cursor
 13      FOR cli_rec IN clientphone LOOP
 14          -- Concatenate names, making sure to avoid an extra comma at the end
 15          IF names IS NULL OR names = '' THEN
 16              names := cli_rec.name;  -- Don't add a comma for the first name
 17          ELSE
 18              names := names || ',' || cli_rec.name;  -- Add comma before subsequent names
 19          END IF;
 20      END LOOP;
 21
 22      -- Return the concatenated names
 23      RETURN names;
 24  END;
